  <script>
    const snapWrapper = document.getElementById('snap-wrapper');
    const sections = Array.from(document.querySelectorAll('.hero-section-view'));
    let currentSectionIndex = 0;
    let isTransitioningCard = false;
    const totalSections = sections.length;

    // Initialize all deck cards with pure GSAP
    function initCardDeck() {
      if (typeof gsap === 'undefined') return;
      sections.forEach((sec, idx) => {
        if (idx === 0) {
          gsap.set(sec, { yPercent: 0, y: 0, scale: 1, opacity: 1, zIndex: 10, visibility: 'visible' });
        } else {
          gsap.set(sec, { yPercent: 100, y: 0, scale: 1, opacity: 1, zIndex: (idx + 1) * 10, visibility: 'visible' });
        }
      });
    }

    function animateSection(id) {
      if (typeof gsap === 'undefined') return;

      if (id === 'sec-0') {
        gsap.fromTo('#sec-0 .bg-slide-img', 
          { yPercent: -8, scale: 1.05, opacity: 0.6 }, 
          { yPercent: 0, scale: 1, opacity: 1, duration: 2.0, ease: 'power3.out' }
        );
        gsap.fromTo('#sec-0 .card-hero-center',
          { y: 25, opacity: 0 },
          { y: 0, opacity: 1, duration: 1.8, ease: 'power3.out', delay: 0.1 }

        gsap.fromTo('.s1-item',

        );
      } else if (id === 'sec-1') {
        // Section 2: Text top-right -> image swipes in from right (slow butter flow)

    // Card Stack Transition
    function goToSection(targetIndex) {
      if (isTransitioningCard) return;
      if (targetIndex < 0 || targetIndex >= totalSections) return;
      if (targetIndex === currentSectionIndex) return;

      isTransitioningCard = true;
      const oldIndex = currentSectionIndex;
      currentSectionIndex = targetIndex;

      // Auto-safety unlock timeout in case onComplete is delayed
      const unlockTimer = setTimeout(() => {
        isTransitioningCard = false;
      }, 1200);

      if (targetIndex > oldIndex) {
        // Stacking forward: incoming card slides UP from yPercent: 100 to yPercent: 0
        const incomingCard = sections[targetIndex];
        gsap.set(incomingCard, { zIndex: (targetIndex + 1) * 10, visibility: 'visible' });

        // Ensure any cards between old and target are also properly positioned
        for (let i = oldIndex + 1; i < targetIndex; i++) {
          gsap.set(sections[i], { yPercent: 0, y: 0, scale: 0.94, opacity: 0.65, zIndex: (i + 1) * 10, visibility: 'visible' });
        }

        // Subtly scale and dim previous card underneath
        gsap.to(sections[oldIndex], {
          scale: 0.94,
          opacity: 0.65,
          duration: 1.0,
          ease: 'power2.out'
        });

        // Glide incoming card UP over deck
        gsap.fromTo(incomingCard, 
          { yPercent: 100, y: 0, scale: 1, opacity: 1 },
          { 
            yPercent: 0, 
            y: 0, 
            scale: 1, 
            opacity: 1, 
            duration: 1.05, 
            ease: 'power3.out',
            onComplete: () => {
              clearTimeout(unlockTimer);
              isTransitioningCard = false;
            }
          }
        );

        animateSection(incomingCard.id);
      } else {
        // Unstacking backward: slide top cards down off the screen
        const targetCard = sections[targetIndex];

        // Restore target card to full scale and opacity
        gsap.to(targetCard, {
          yPercent: 0,
          scale: 1,
          opacity: 1,
          duration: 0.9,
          ease: 'power2.out'
        });

        let completed = 0;
        const totalToDismiss = oldIndex - targetIndex;

        for (let i = oldIndex; i > targetIndex; i--) {
          const cardToDismiss = sections[i];
          gsap.to(cardToDismiss, {
            yPercent: 100,
            y: 0,
            scale: 1,
            opacity: 1,
            duration: 0.9,
            ease: 'power2.inOut',
            onComplete: () => {
              completed++;
              if (completed >= totalToDismiss) {
                clearTimeout(unlockTimer);
                isTransitioningCard = false;
              }
            }
          });
        }

        animateSection(targetCard.id);
      }
    }

    // Wheel Scroll Event
    let lastWheelTime = 0;
    window.addEventListener('wheel', (e) => {
      if ((voteModal && voteModal.open) || (scheduleModal && scheduleModal.open)) return;
      if (Math.abs(e.deltaY) < 15) return;
      const now = Date.now();
      if (now - lastWheelTime < 650 || isTransitioningCard) return;

      if (e.deltaY > 0) {
        if (currentSectionIndex < totalSections - 1) {
          lastWheelTime = now;
          goToSection(currentSectionIndex + 1);
        }
      } else if (e.deltaY < 0) {
        if (currentSectionIndex > 0) {
          lastWheelTime = now;
          goToSection(currentSectionIndex - 1);
        }
      }
    }, { passive: true });

    // Touch Swipe Event for Mobile & Tablets

    let touchStartX = 0;
    window.addEventListener('touchstart', (e) => {
      if ((voteModal && voteModal.open) || (scheduleModal && scheduleModal.open)) return;
      touchStartY = e.touches[0].clientY;
      touchStartX = e.touches[0].clientX;
    }, { passive: true });

    window.addEventListener('touchend', (e) => {
      if ((voteModal && voteModal.open) || (scheduleModal && scheduleModal.open)) return;
      const touchEndY = e.changedTouches[0].clientY;
      const touchEndX = e.changedTouches[0].clientX;
      const diffY = touchStartY - touchEndY;
      const diffX = touchStartX - touchEndX;

      if (Math.abs(diffY) > 35 && Math.abs(diffY) > Math.abs(diffX)) {
        if (diffY > 0 && currentSectionIndex < totalSections - 1) {
          goToSection(currentSectionIndex + 1);
        } else if (diffY < 0 && currentSectionIndex > 0) {
          goToSection(currentSectionIndex - 1);
        }
      }
    }, { passive: true });

    // Keyboard Navigation

      if ((voteModal && voteModal.open) || (scheduleModal && scheduleModal.open)) return;
      if (['ArrowDown', 'PageDown', ' '].includes(e.key)) {
        e.preventDefault();
        if (currentSectionIndex < totalSections - 1) goToSection(currentSectionIndex + 1);
      } else if (['ArrowUp', 'PageUp'].includes(e.key)) {
        e.preventDefault();
        if (currentSectionIndex > 0) goToSection(currentSectionIndex - 1);

    function initPetalCanvas() {
      const canvas = document.getElementById('petal-canvas');
      if (!canvas) return;
      const ctx = canvas.getContext('2d');

      let width = canvas.width = window.innerWidth;
      let height = canvas.height = window.innerHeight;

      window.addEventListener('resize', () => {
        width = canvas.width = window.innerWidth;
        height = canvas.height = window.innerHeight;
      });

      const petals = [];
      const petalColors = [
        'rgba(254, 215, 226, 0.45)',
        'rgba(254, 243, 199, 0.45)',
        'rgba(253, 230, 138, 0.35)',
        'rgba(255, 255, 255, 0.55)'
      ];

      for (let i = 0; i < 20; i++) {
        petals.push({
          x: Math.random() * width,
          y: Math.random() * height,
          size: Math.random() * 6 + 3,
          speedY: Math.random() * 0.6 + 0.25,
          speedX: (Math.random() - 0.5) * 0.4,
          angle: Math.random() * Math.PI * 2,
          angularSpeed: (Math.random() - 0.5) * 0.02,
          color: petalColors[Math.floor(Math.random() * petalColors.length)]
        });
      }

      function animate() {
        ctx.clearRect(0, 0, width, height);

        petals.forEach(p => {
          p.y += p.speedY;
          p.x += Math.sin(p.angle) * 0.5 + p.speedX;
          p.angle += p.angularSpeed;

          if (p.y > height + 15) {
            p.y = -15;
            p.x = Math.random() * width;
          }

          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate(p.angle);
          ctx.fillStyle = p.color;
          ctx.beginPath();
          ctx.ellipse(0, 0, p.size, p.size * 0.6, 0, 0, Math.PI * 2);
          ctx.fill();
          ctx.restore();
        });

        requestAnimationFrame(animate);
      }

      animate();
    }
